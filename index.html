function renderMiniCalendar() {

const daysContainer = document.getElementById('mini-cal-days');

daysContainer.innerHTML = '';

const dateObj = new Date(miniCalYear, miniCalMonth, day);

if (day === today.getDate() && miniCalMonth === today.getMonth() && miniCalYear === today.getFullYear()) {

cell.classList.add('today');

const holidayInfo = getHolidayInfo(dateStr, dateObj);

if (holidayInfo.isHoliday) cell.classList.add('is-holiday');

const isStart = dateStr === modalRangeStart;

const isEnd = modalRangeEnd ? dateStr === modalRangeEnd : isStart;

const inRange = isDateInSelectedRange(dateStr);

if (inRange) cell.classList.add('in-range');

if (isStart) cell.classList.add('range-start');

if (isEnd) cell.classList.add('range-end');

cell.innerHTML = `<span class="mini-cal-num">${day}</span>`;

cell.onclick = () => handleMiniCalClick(dateStr);

daysContainer.appendChild(cell);

}

}

function isDateInSelectedRange(dateStr) {

if (!modalRangeStart) return false;

if (!modalRangeEnd) return dateStr === modalRangeStart;

return dateStr >= modalRangeStart && dateStr <= modalRangeEnd;

}

function handleMiniCalClick(dateStr) {

modalRangeEnd = null;

selectedDatesList = [dateStr];

renderMiniCalendar();

updateDateSummary();

}

if (!modalRangeStart || modalRangeEnd) {

modalRangeStart = dateStr;

modalRangeEnd = null;

modalRangeEnd = null;

modalRangeEnd = modalRangeStart;

modalRangeStart = dateStr;

modalRangeEnd = dateStr;

}

} else {

selectedDatesList = [modalRangeStart];

}

renderMiniCalendar();

updateDateSummary();

}

function getCalculatedDuration() {

const startTime = document.getElementById('start-time').value;

const endTime = document.getElementById('end-time').value;

if (!startTime || !endTime) return 0;

const [sHours, sMins] = startTime.split(':').map(Number);

const [eHours, eMins] = endTime.split(':').map(Number);

let startTotalMins = sHours * 60 + sMins;

let endTotalMins = eHours * 60 + eMins;

if (endTotalMins < startTotalMins) {

endTotalMins += 24 * 60;

}

let diffMins = endTotalMins - startTotalMins;

if (document.getElementById('no-lunch-check').checked) {

const deductMins = parseFloat(document.getElementById('deduct-minutes').value) || 0;

diffMins = Math.max(0, diffMins - deductMins);

}

return Math.round((diffMins / 60) * 100) / 100;

}

async function shareOrDownloadFile(filename, mimeType, content) {

try {

const blob = new Blob([content], { type: mimeType });

const file = new File([blob], filename, { type: mimeType });

if (navigator.canShare && navigator.canShare({ files: [file] })) {

await navigator.share({

files: [file],

title: filename,

text: '兼職紀錄備份'

});

return;

}

const url = URL.createObjectURL(blob);

const link = document.createElement('a');

link.href = url;

link.download = filename;

document.body.appendChild(link);

link.click();

document.body.removeChild(link);

setTimeout(() => URL.revokeObjectURL(url), 2000);

} catch (err) {

if (err && err.name === 'AbortError') return;

console.error('Export Error:', err);

alert('匯出失敗');

}

}

if (!file) return;

const reader = new FileReader();

reader.onload = function(e) {

try {

if (Array.isArray(importedData)) {

shifts = importedData;

alert('匯入成功！');

} else { alert('無效的備份檔案'); }

} catch (err) { alert('解析檔案失敗'); }

};

reader.readAsText(file);

event.target.value = '';

}

/* Render Calendar */

function renderCalendar() {

autoUpdatePastPlannedShifts();

document.getElementById('calendar-month-year').textContent = `${currentYear} 年 ${currentMonth + 1} 月`;

const daysContainer = document.getElementById('calendar-days');

daysContainer.innerHTML = '';

let rawFirstDay = new Date(currentYear, currentMonth, 1).getDay();

let firstDay = (rawFirstDay === 0) ? 6 : rawFirstDay - 1;

const totalDays = new Date(currentYear, currentMonth + 1, 0).getDate();

const prevLastDay = new Date(currentYear, currentMonth, 0).getDate();

const today = new Date();

for (let i = firstDay - 1; i >= 0; i--) {

const cell = document.createElement('div');

cell.className = 'day-cell other-month';

cell.innerHTML = `<span class="day-number">${prevLastDay - i}</span>`;

daysContainer.appendChild(cell);

}

let mPayCompleted = 0;

let mPayTotal = 0;

for (let day = 1; day <= totalDays; day++) {

const cell = document.createElement('div');

cell.className = 'day-cell';

const dateStr = `${currentYear}-${String(currentMonth + 1).padStart(2, '0')}-${String(day).padStart(2, '0')}`;

const dateObj = new Date(currentYear, currentMonth, day);

const holidayInfo = getHolidayInfo(dateStr, dateObj);

cell.onclick = () => {

openModal(dateStr);

};

if (day === today.getDate() && currentMonth === today.getMonth() && currentYear === today.getFullYear()) {

cell.classList.add('today');

}

if (holidayInfo.isHoliday) {

cell.classList.add('is-holiday');

}

let html = `<span class="day-number">${day}</span>`;

if (holidayInfo.isHoliday && holidayInfo.label) {

html += `<div class="holiday-label">${holidayInfo.label}</div>`;

}

const dayShifts = shifts.filter(s => s.date === dateStr);

let shiftsHtml = '';

dayShifts.forEach(s => {

const pay = s.rate * s.duration;

mPayTotal += pay;

if (s.type === 'completed' || !s.type) {

mPayCompleted += pay;

}

const isPlanned = s.type === 'planned';

const timeTag = `${formatShortTime(s.startTime)}-${formatShortTime(s.endTime)}`;

shiftsHtml += `

<div class="shift-dot-item ${isPlanned ? 'planned' : ''}" onclick="event.stopPropagation(); openShiftDetail(${s.id})" title="${s.company} (${timeTag}) $${pay}">

<div class="comp-name">${s.company}</div>

<div class="shift-time">${timeTag}</div>

<div class="shift-money">$${Math.round(pay)}</div>

</div>

`;

});

html += `<div class="shift-items-wrap">${shiftsHtml}</div>`;

cell.innerHTML = html;

daysContainer.appendChild(cell);

}

const totalRendered = firstDay + totalDays;

const nextDays = (7 - (totalRendered % 7)) % 7;

for (let i = 1; i <= nextDays; i++) {

const cell = document.createElement('div');

cell.className = 'day-cell other-month';

cell.innerHTML = `<span class="day-number">${i}</span>`;

daysContainer.appendChild(cell);

}

document.getElementById('month-pay').textContent = `$ ${Math.round(mPayCompleted).toLocaleString()}`;

document.getElementById('month-total-est').textContent = `$ ${Math.round(mPayTotal).toLocaleString()}`;

}

function renderAnalytics() {

autoUpdatePastPlannedShifts();

updateMonthSalaryCard();

document.getElementById('analytics-month-year').textContent = `${currentYear} 年 ${currentMonth + 1} 月`;

const monthPrefix = `${currentYear}-${String(currentMonth + 1).padStart(2, '0')}`;

const monthShifts = shifts.filter(s => s.date && s.date.startsWith(monthPrefix));

let totalHours = 0;

let totalPay = 0;

const companyMap = {};

monthShifts.forEach(s => {

const pay = s.rate * s.duration;

if (s.type === 'completed' || !s.type) {

totalHours += s.duration;

totalPay += pay;

}

if (!companyMap[s.company]) {

companyMap[s.company] = { hours: 0, pay: 0, totalEstPay: 0, count: 0 };

}

companyMap[s.company].totalEstPay += pay;

if (s.type === 'completed' || !s.type) {

companyMap[s.company].hours += s.duration;

companyMap[s.company].pay += pay;

companyMap[s.company].count += 1;

}

});

const avgRate = totalHours > 0 ? (totalPay / totalHours) : 0;

document.getElementById('stat-hours').textContent = `${Math.round(totalHours * 10) / 10}h`;

document.getElementById('stat-total-pay').textContent = `$${Math.round(totalPay).toLocaleString()}`;

const companyBreakdownContainer = document.getElementById('company-breakdown-container');

const companyEntries = Object.entries(companyMap).sort((a, b) => b[1].totalEstPay - a[1].totalEstPay);

const allCompanyNames = [...new Set(shifts.map(s => s.company))];

const today = new Date();

const currentDayNum = today.getDate();

const isCurrentViewingMonth = (today.getFullYear() === currentYear && today.getMonth() === currentMonth);

if (companyEntries.length === 0) {

companyBreakdownContainer.innerHTML = `<div class="empty-msg">本月沒有工時紀錄</div>`;

} else {

companyBreakdownContainer.innerHTML = companyEntries.map(([name, data]) => {

const idx = allCompanyNames.indexOf(name);

const color = companyColorPalette[idx % companyColorPalette.length];

const pct = totalPay > 0 ? Math.round((data.pay / totalPay) * 100) : 0;

const avgCompRate = data.hours > 0 ? Math.round(data.pay / data.hours) : 0;

const pDay = companyPaydays[name] || 7; // 預設 7 號出糧

let paydayStatusHtml = '';

return `

<div class="company-stat-item">

<div class="company-stat-name">

<span class="company-stat-dot" style="background:${color.text};"></span>

${name}

</div>

<div class="company-stat-pay">$${Math.round(data.totalEstPay).toLocaleString()}</div>

<div class="company-stat-meta">

<span>${data.count} job • ${Math.round(data.hours * 10) / 10}h ($${avgCompRate}/hr)</span>

${paydayStatusHtml}

</div>

</div>

`;

}).join('');

}
